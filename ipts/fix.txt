import React, { useEffect, useState } from 'react';
import { FileText, Plus, CheckCircle, Clock, AlertTriangle, ExternalLink, Code2, MessageSquare, Star, ArrowRight, X, Save, Trash2, RefreshCw } from 'lucide-react';
import { ProgressReport, Internship } from '../../types';
import { api } from '../../services/api';

export const WeeklyReports: React.FC = () => {
  const [reports, setReports] = useState<ProgressReport[]>([]);
  const [internships, setInternships] = useState<Internship[]>([]);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);

  const [selectedInternshipId, setSelectedInternshipId] = useState<number | null>(null);
  const [selectedReportId, setSelectedReportId] = useState<number | null>(null);
  const [activeReport, setActiveReport] = useState<ProgressReport | null>(null);

  const [formData, setFormData] = useState({
    week_number: 1,
    tasks_completed: '',
    technical_skills: '',
    learning_outcomes: '',
    challenges_faced: '',
    solutions_implemented: '',
    project_progress: '',
    github_link: '',
    demo_link: '',
    next_week_plan: '',
    student_reflection: '',
    hours_worked: 0,
    proof_links: [] as string[],
  });

  const [feedbackForm, setFeedbackForm] = useState({
    score: 8,
    comments: '',
    decision: 'APPROVED' as 'APPROVED' | 'NEEDS_IMPROVEMENT',
    improvement_task: '',
  });

  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);

  const fetchReports = async () => {
    setLoading(true);
    try {
      // Check if user is authenticated first
      const userData = JSON.parse(localStorage.getItem('user_data') || '{}');
      if (!userData.id) {
        setReports([]);
        setInternships[];
        setLoading(false);
        return;
      }
      const data = await api.get<ProgressReport[] | { results: ProgressReport[] }>('/reports/');
      const list = Array.isArray(data) ? data : data.results || [];
      setReports(list);
    } catch (err: any) {
      console.error('Failed to fetch reports:', err);
      setMessage({ type: 'error', text: 'Failed to load reports' });
    }
    try {
      const data = await api.get<Internship[] | { results: Internship[] }>('/internships/');
      const list = Array.isArray(data) ? data : data.results || [];
      setInternships(list);
      const active = list.find((i: Internship) => i.status === 'ACTIVE');
      if (active) setSelectedInternshipId(active.id);
    } catch (err: any) {
      console.error('Failed to fetch internships:', err);
    } finally { setLoading(false); }
  };

  useEffect(() => { fetchReports(); }, []);
  const getActiveInternship = () => internships.find((i: Internship) => i.id === selectedInternshipId) || null;
  const activeInternship = getActiveInternship();

  const handleInputChange = (field: string, value: any) => {
    setFormData(prev => ({ ...prev, [field]: value }));
  };

  const handleAddProofLink = () => {
    if (formData.proof_links.length < 3) { setFormData(prev => ({ ...prev, proof_links: [...prev.proof_links, ''] })); }
  };

  const handleRemoveProofLink = (index: number) => {
    setFormData(prev => ({ ...prev, proof_links: prev.proof_links.filter((_, i) => i !== index) }));
  };

  const handleProofLinkChange = (index: number, value: string) => {
    setFormData(prev => { const links = [...prev.proof_links]; links[index] = value; return { ...prev, proof_links: links }; });
  };

  const validateRequired = () => {
    if (!activeInternship) { setMessage({ type: 'error', text: 'Please select your active internship first.' }); return false; }
    if (!formData.tasks_completed.trim()) { setMessage({ type: 'error', text: 'Tasks Completed is required.' }); return false; }
    if (!formData.technical_skills.trim()) { setMessage({ type: 'error', text: 'Technical Skills are required.' }); return false; }
    if (!formData.learning_outcomes.trim()) { setMessage({ type: 'error', text: 'Learning Outcomes are required.' }); return false; }
    const hours = Number(formData.hours_worked);
    if (isNaN(hours) || hours < 0 || hours > 80) { setMessage({ type: 'error', text: 'Hours worked must be between 0 and 80.' }); return false; }
    const links = formData.proof_links.filter(l => l.trim());
    if (links.length > 0) { for (const link of links) { if (!link.startsWith('http://') && !link.startsWith('https://')) { setMessage({ type: 'error', text: 'Invalid link: ' + link + '. Must be HTTP/HTTPS.' }); return false; } } }
    return true;
  };

  const handleSubmitReport = async (status: 'DRAFT' | 'SUBMITTED') => {
    if (!validateRequired()) return;
    setSaving(true);
    setMessage(null);
    try {
      const payload = {
        internship: selectedInternshipId!,
        week_number: formData.week_number,
        tasks_completed: formData.tasks_completed,
        technical_skills: formData.technical_skills,
        learning_outcomes: formData.learning_outcomes,
        challenges_faced: formData.challenges_faced,
        solutions_implemented: formData.solutions_implemented,
        project_progress: formData.project_progress,
        github_link: formData.github_link,
        demo_link: formData.demo_link,
        next_week_plan: formData.next_week_plan,
        student_reflection: formData.student_reflection,
        hours_worked: Number(formData.hours_worked),
        proof_links: formData.proof_links.filter(l => l.trim()),
        status,
      };
      if (status === 'SUBMITTED') {
        await api.post('/reports/', payload);
      } else {
        await api.post('/reports/', payload);
      }
      fetchReports();
    } catch (err) {
      console.error(err);
    } finally {
      setSaving(false);
    }
  };

  const handleResubmit = async () => {
    if (!activeReport) return;
    if (activeReport.status === 'SUBMITTED') {
      setMessage({ type: 'error', text: 'This report is already submitted for review.' });
      return;
    }
    if (activeReport.status === 'APPROVED') {
      setMessage({ type: 'error', text: 'Approved reports cannot be edited.' });
      return;
    }
    if (activeReport.status !== 'NEEDS_IMPROVEMENT') {
      setMessage({ type: 'error', text: 'Only reports marked NEEDS_IMPROVEMENT can be resubmitted.' });
      return;
    }
    if (!formData.tasks_completed.trim() || !formData.technical_skills.trim() || !formData.learning_outcomes.trim()) {
      setMessage({ type: 'error', text: 'All required fields must be filled.' });
      return;
    }
    try {
      await api.patch('/reports/' + activeReport.id + '/', {
        tasks_completed: formData.tasks_completed,
        technical_skills: formData.technical_skills,
        learning_outcomes: formData.learning_outcomes,
        challenges_faced: formData.challenges_faced,
        solutions_implemented: formData.solutions_implemented,
        project_progress: formData.project_progress,
        github_link: formData.github_link,
        demo_link: formData.demo_link,
        next_week_plan: formData.next_week_plan,
        student_reflection: formData.student_reflection,
        hours_worked: Number(formData.hours_worked),
        proof_links: formData.proof_links.filter(l => l.trim()),
        status: 'SUBMITTED',
      });
      setMessage({ type: 'success', text: 'Report resubmitted for review.' });
      setActiveReport(null);
      fetchReports();
    } catch (err) {
      setMessage({ type: 'error', text: (err instanceof Error ? err.message : String(err)) || 'Failed to resubmit report' });
    }
  };

  const handleDeleteReport = async (id: number) => {
    if (!window.confirm('Are you sure you want to delete this draft report?')) return;
    try {
      await api.delete('/reports/' + id + '/');
      setMessage({ type: 'success', text: 'Report deleted.' });
      fetchReports();
    } catch (err) {
      setMessage({ type: 'error', text: (err instanceof Error ? err.message : String(err)) || 'Failed to delete report' });
    }
  };

  const openReviewModal = (report: ProgressReport) => {
    setActiveReport(report);
    setFeedbackForm({ score: 8, comments: '', decision: 'APPROVED', improvement_task: '' });
  };

  const handleFeedbackSubmit = async (decision: 'APPROVED' | 'NEEDS_IMPROVEMENT') => {
    if (!activeReport) return;
    if (decision === 'NEEDS_IMPROVEMENT' && !feedbackForm.comments.trim()) {
      setMessage({ type: 'error', text: 'Please provide feedback comments.' });
      return;
    }
    setSaving(true);
    try {
      await api.post('/reports/' + activeReport.id + '/feedback/', {
        score: feedbackForm.score,
        comments: feedbackForm.comments,
        decision,
        task_title: feedbackForm.improvement_task,
      });
      setMessage({ type: 'success', text: 'Report ' + (decision === 'APPROVED' ? 'approved' : 'marked for improvement.') });
      setActiveReport(null);
      fetchReports();
    } catch (err) {
      setMessage({ type: 'error', text: (err instanceof Error ? err.message : String(err)) || 'Failed to submit feedback' });
    } finally {
      setSaving(false);
    }
  };

  const getStatusBadge = (status: string) => {
    switch (status) {
      case 'APPROVED': return <span className='badge badge-success'>Approved</span>;
      case 'NEEDS_IMPROVEMENT': return <span className='badge badge-warning'>Needs Improvement</span>;
      case 'SUBMITTED': return <span className='badge badge-published'>Submitted</span>;
      case 'UNDER_REVIEW': return <span className='badge badge-info'>Under Review</span>;
      case 'DRAFT': return <span className='badge badge-secondary'>Draft</span>;
      default: return <span className='badge badge-secondary'>{status}</span>;
    }
  };

  const formatDate = (dateStr: string | null) => {
    if (!dateStr) return '-';
    try { return new Date(dateStr).toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' }); }
    catch { return dateStr; }
  };

  return (
    <div>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '1.5rem' }}>
        <div>
          <h1 style={{ fontSize: '22px', fontWeight: 700, color: '#0f172a' }}>Weekly Progress Reports</h1>
          <p style={{ color: '#64748b', fontSize: '14px' }}>Track and submit your weekly internship progress reports.</p>
        </div>
        <button onClick={fetchReports} className='btn btn-outline' disabled={loading} style={{ display: 'flex', alignItems: 'center', gap: '6px' }}>
          <RefreshCw size={16} /> Refresh
        </button>
      </div>      {message && (
        <div style={{padding: '12px 16px', borderRadius: '8px', marginBottom: '1rem', backgroundColor: message.type === 'success' ? '#ecfdf5' : '#fef2f2', color: message.type === 'success' ? '#059669' : '#dc2626', fontSize: '14px', fontWeight: 500, border: '1px solid #d1d5db'}}>
          {message.text}
        </div>
      )}
      <div style={{display: 'grid', gridTemplateColumns: 'repeat(auto-fit, minmax(340px, 1fr))', gap: '1.5rem', marginBottom: '2rem'}}>
        <div className='card'>
          <h3 style={{fontSize: '15px', fontWeight: 600, color: '#0f172a', marginBottom: '1rem', display: 'flex', alignItems: 'center', gap: '8px'}}>
            <div style={{display: 'flex', alignItems: 'center'}}>
              <div style={{width: '18px', height: '18px', borderRadius: '50%', backgroundColor: '#2563eb', color: '#fff', display: 'flex', alignItems: 'center', justifyContent: 'center'}}>
                <svg width='12' height='12' viewBox='0 0 24 24' fill='none' stroke='currentColor' strokeWidth='2' strokeLinecap='round' strokeLinejoin='round'><rect x='3' y='3' width='18' height='18' rx='2' /></svg>
              </div>
            </div>
            <span>Active Internship</span>
          </h3>
          {activeInternship ? (
            <div style={{color: '#94a3b8', fontSize: '13px', textAlign: 'center', padding: '20px'}}>No active internship available. Complete internship setup first.</div>
          ) : (
            <div style={{color: '#94a3b8', fontSize: '13px', textAlign: 'center', padding: '20px'}}>No active internship available. Complete internship setup first.</div>
          )}
        </div>
        <div className='card'>
          <h3 style={{fontSize: '15px', fontWeight: 600, color: '#0f172a', marginBottom: '1rem', display: 'flex', alignItems: 'center', gap: '8px'}}>
            <div style={{display: 'flex', alignItems: 'center'}}>
              <div style={{width: '18px', height: '18px', borderRadius: '50%', backgroundColor: '#2563eb', color: '#fff', display: 'flex', alignItems: 'center', justifyContent: 'center'}}>
                <svg width='12' height='12' viewBox='0 0 24 24' fill='none' stroke='currentColor' strokeWidth='2' strokeLinecap='round' strokeLinejoin='round'><line x1='12' y1='5' x2='12' y2='19' /><line x1='5' y1='12' x2='19' y2='12' /></svg>
              </div>
            </div>
            <span>New Report #{formData.week_number}</span>
          </h3>
          <div style={{display: 'flex', flexDirection: 'column', gap: '10px'}}>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Tasks Completed *</label>
              <textarea className='form-control' rows={3} placeholder='What did you complete this week?' value={formData.tasks_completed} onChange={(e) => handleInputChange('tasks_completed', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Technical Skills *</label>
              <textarea className='form-control' rows={2} placeholder='Tools, frameworks, techniques learned' value={formData.technical_skills} onChange={(e) => handleInputChange('technical_skills', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Learning Outcomes *</label>
              <textarea className='form-control' rows={2} placeholder='Key concepts and skills gained' value={formData.learning_outcomes} onChange={(e) => handleInputChange('learning_outcomes', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Challenges Faced</label>
              <textarea className='form-control' rows={2} placeholder='What obstacles did you encounter?' value={formData.challenges_faced} onChange={(e) => handleInputChange('challenges_faced', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Solutions Implemented</label>
              <textarea className='form-control' rows={2} placeholder='How did you resolve them?' value={formData.solutions_implemented} onChange={(e) => handleInputChange('solutions_implemented', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Project Progress</label>
              <textarea className='form-control' rows={2} placeholder='Status update on current projects' value={formData.project_progress} onChange={(e) => handleInputChange('project_progress', e.target.value)} />
            </div>
            <div style={{display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '8px'}}>
              <div>
                <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>GitHub Link</label>
                <input type='url' className='form-control' placeholder='https://github.com/...' value={formData.github_link} onChange={(e) => handleInputChange('github_link', e.target.value)} />
              </div>
              <div>
                <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Demo Link</label>
                <input type='url' className='form-control' placeholder='https://...' value={formData.demo_link} onChange={(e) => handleInputChange('demo_link', e.target.value)} />
              </div>
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Next Week Plan</label>
              <textarea className='form-control' rows={2} placeholder='Plans for the upcoming week' value={formData.next_week_plan} onChange={(e) => handleInputChange('next_week_plan', e.target.value)} />
            </div>
            <div>
              <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Student Reflection</label>
              <textarea className='form-control' rows={3} placeholder='What went well? What to improve?' value={formData.student_reflection} onChange={(e) => handleInputChange('student_reflection', e.target.value)} />
            </div>
            <div style={{display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '8px'}}>
              <div>
                <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Hours Worked (0-80)</label>
                <input type='number' min='0' max='80' className='form-control' value={formData.hours_worked} onChange={(e) => handleInputChange('hours_worked', e.target.value)} />
              </div>
              <div>
                <label style={{fontSize: '13px', fontWeight: 500, color: '#475569', display: 'block', marginBottom: '4px'}}>Proof Links (max 3)</label>
                <div style={{display: 'flex', flexWrap: 'wrap', gap: '6px', marginTop: '4px'}}>
                  {formData.proof_links.map((link, i) => (
                    <div key={i} style={{display: 'flex', gap: '4px', alignItems: 'center'}}>
                      <input type='url' className='form-control' style={{width: '160px'}} placeholder='https://...' value={link} onChange={(e) => handleProofLinkChange(i, e.target.value)} />
                      <button type='button' onClick={() => handleRemoveProofLink(i)} style={{cursor: 'pointer', color: '#ef4444', fontSize: '16px', padding: 0, border: 'none', background: 'none'}}>&times;</button>
                    </div>
                  ))}
                  <button type='button' onClick={handleAddProofLink} className='btn btn-outline' style={{padding: '4px 10px', fontSize: '12px'}} disabled={formData.proof_links.length >= 3}>+ Add</button>
                </div>
              </div>
            </div>
          </div>
          <div style={{display: 'flex', gap: '0.75rem', marginTop: '1rem'}}>
            <button onClick={() => handleSubmitReport('DRAFT')} className='btn btn-outline' style={{display: 'flex', alignItems: 'center', gap: '6px'}}>
              <Save size={16} /> Save Draft
            </button>
            <button onClick={() => handleSubmitReport('SUBMITTED')} className='btn btn-primary' style={{display: 'flex', alignItems: 'center', gap: '6px'}} disabled={saving}>
              <CheckCircle size={16} /> Submit for Review
            </button>
          </div>
        </div>
      </div>
      {activeReport && (
        <div style={{position: 'fixed', top: 0, left: 0, right: 0, bottom: 0, backgroundColor: 'rgba(0,0,0,0.5)', display: 'flex', alignItems: 'center', justifyContent: 'center', zIndex: 1000}}>
          <div style={{position: "relative", width: "100%", maxWidth: "600px", backgroundColor: "#fff", borderRadius: "12px", boxShadow: "0 25px 50px rgba(0,0,0,0.25)", maxHeight: "90vh", overflow: "auto"}}>
            <div style={{display: "flex", justifyContent: "space-between", alignItems: "center", padding: "1.25rem 1.5rem", borderBottom: "1px solid #e2e8f0"}}>
              <h3 style={{fontSize: "17px", fontWeight: 700, color: "#0f172a"}}>Review Student Report</h3>
              <button onClick={() => setActiveReport(null)} style={{cursor: "pointer", background: "none", border: "none", color: "#64748b", padding: 4}}><X size={20} /></button>
            </div>
            <div style={{padding: "1.5rem"}}>
              <div style={{display: "grid", gridTemplateColumns: "1fr 1fr", gap: "1rem", marginBottom: "1.5rem"}}>
                <div><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "4px"}}>Student Name</label><div style={{backgroundColor: "#f8faf9", padding: "10px", borderRadius: "8px", fontSize: "14px", fontWeight: 500}}>{activeReport.student_name}</div></div>
                <div><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "4px"}}>Organization</label><div style={{backgroundColor: "#f8faf9", padding: "10px", borderRadius: "8px", fontSize: "14px", fontWeight: 500}}>{activeReport.organization_name}</div></div>
                <div><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "4px"}}>Role Title</label><div style={{backgroundColor: "#f8faf9", padding: "10px", borderRadius: "8px", fontSize: "14px", fontWeight: 500}}>{activeReport.role_title}</div></div>
                <div><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "4px"}}>Week #{activeReport.week_number}</label><div style={{backgroundColor: "#f8faf9", padding: "10px", borderRadius: "8px", fontSize: "14px", fontWeight: 500}}>{activeReport.week_number}</div></div>
              </div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Tasks Completed</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.tasks_completed}</p></div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Technical Skills</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.technical_skills}</p></div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Learning Outcomes</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.learning_outcomes}</p></div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Challenges Faced</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.challenges_faced}</p></div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Solutions Implemented</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.solutions_implemented}</p></div>
              <div style={{marginBottom: "1.5rem"}}><label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Project Progress</label><p style={{margin: 0, whiteSpace: "pre-wrap", lineHeight: 1.7, color: "#1e293b"}}>{activeReport.project_progress}</p></div>
              <div style={{marginBottom: "1.5rem"}}>
                <label style={{fontSize: "13px", fontWeight: 500, color: "#475569", display: "block", marginBottom: "6px"}}>Proof Links</label>
                <div style={{display: "flex", flexWrap: "wrap", gap: "8px", marginTop: "6px"}}>
                  {activeReport.proof_links?.map((link: string, i: number) => (
                    <a key={i} href={link} target="_blank" rel="noopener noreferrer" style={{display: "inline-flex", alignItems: "center", gap: "6px", padding: "6px 12px", backgroundColor: "#eff6ff", color: "#1d4ed8", borderRadius: "6px", textDecoration: "none", fontSize: "13px", fontWeight: 500, border: "1px solid #93c5fd"}}>
                      <ExternalLink size={14} />{link}
                    </a>
                  ))}
                </div>
              </div>
              <div style={{display: "flex", justifyContent: "space-between", alignItems: "center", flexWrap: "wrap", gap: "1rem", marginBottom: "1.5rem"}}>
                <div style={{display: "flex", gap: "0.75rem"}}>
                  <button onClick={() => setFeedbackForm({score: 8, comments: "", decision: "APPROVED", improvement_task: ""})} className="btn btn-outline" style={{display: "flex", alignItems: "center", gap: "6px"}}><Star size={16} /> Mark Reviewed</button>
                </div>
                <div style={{display: "flex", gap: "0.75rem"}}>
                  <button onClick={() => handleDeleteReport(activeReport.id)} className="btn btn-danger" style={{display: "flex", alignItems: "center", gap: "6px"}}><Trash2 size={16} /> Delete</button>
                </div>
              </div>
              {activeReport.status !== "DRAFT" && (
                <div style={{padding: "10px 14px", backgroundColor: "#eff6ff", borderRadius: "8px", border: "1px solid #93c5fd", fontSize: "13px", color: "#1d4ed8", marginBottom: "1.5rem"}}>
                  Current Status: <strong>{activeReport.status}</strong>
                  {activeReport.coordinator_comment && <div style={{marginTop: "8px"}}>Coordinator: <strong>{activeReport.coordinator_comment}</strong></div>}
                  {activeReport.reviewed_by_name && <div style={{marginTop: "4px"}}>Reviewed by: <strong>{activeReport.reviewed_by_name}</strong></div>}
                </div>
              )}
              <div style={{display: "flex", justifyContent: "flex-end", padding: "1rem 1.5rem", borderTop: "1px solid #e2e8f0", gap: "0.75rem"}}>

                <button onClick={() => setActiveReport(null)} className="btn btn-outline" disabled={saving}>Cancel</button>
                <button onClick={() => handleFeedbackSubmit(activeReport.status === "APPROVED" ? "APPROVED" : "NEEDS_IMPROVEMENT")} className="btn btn-primary" disabled={saving} style={{display: "flex", alignItems: "center", gap: "6px"}}><CheckCircle size={16} /> Submit Review</button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
